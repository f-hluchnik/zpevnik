\song{Vodník}\artist{nedvedi}

\vers[1]{
\chord{D}Posvěcený deštěm, který padá z nebe,\\
\chord{A}vlasy jako vodník, pod šosem to zebe,\\
z kůže \chord{Em}břízy když narodil se \chord{A\7}oheň,\\
\chord{D}vysuší jim \chord{Hm}promáčenej \chord{A\7}den.
}

\vers[2]{
Malovaný sluncem, hnědou barvou klasů,\\
indiánský hrátky starejch dobrejch časů,\\
večer zvedaj' hlavu do rozsvícenejch hvězd,\\
lesem zazní písně dlouhejch cest.
}

\vers[3]{
Na krku svý znamení: raka, kotvu, list,\\
na rukávech domovenky, z kterejch můžeš číst,\\
na srdci sem-tam záplata, jak život šel,\\
mlčej' k ohni, když se rozhořel.
}

\chorus{
Krásný \chord{D}noci plný bílejch hvězd a kamará\chord{A}dů,\\
\chord{D}vyzpívaný touhy kluků z měst, plný bláznivejch \chord{A}snů.
}

\vers[4]{
Jiný písně zazněj', než učili se zpívat,\\
jinak viděj' svět než učili se dívat,\\
v celý přírodě se nenajde tak hned,\\
tolik hrůzy, co už strořil ten náš svět.
}

\vers[5]{
A tak vyšlápli si ránem a první listy stránek\\
popisujou písmem, co poslal z lesů vánek,\\
hledaj' na slunci si místo, kdo z nich ví,\\
co zítra na ně svět zas vymyslí.
} \cue{R}
