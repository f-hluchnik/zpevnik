% The back of the cover, read by print-cover.tex and proposals.tex (XeLaTeX): the type, the
% colours, the way to the online songbook and the six proposed designs. A design is drawn by
% \backdesign{<name>} inside a TikZ picture whose coordinate T is the trim's lower left corner;
% it covers the bleed too. Nothing here comes from the front's drawing (README.md).
\usepackage{fontspec}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{tikz}
\usepackage{qrcode}
\usepackage{etoolbox}

% The page (mm): A5, and the bleed the printer cuts off.
\newcommand*{\trimwidth}{148}
\newcommand*{\trimheight}{210}
\newcommand*{\bleed}{3}

% Alegreya and FreeSerif (for the lily) come with TeX Live (packages alegreya, gnu-freefont).
\setmainfont{Alegreya}[Extension=.otf, UprightFont=*-Regular, ItalicFont=*-Italic,
  SmallCapsFont=AlegreyaSC-Regular]
\newfontfamily\symbolfont{FreeSerif.otf}

\definecolor{ink}{HTML}{2E261E}       % the text and the QR code
\definecolor{spruce}{HTML}{3F5236}    % the near trees
\definecolor{farspruce}{HTML}{8E9C82} % the far trees, in the haze

% The way to the online songbook: the book's title, the QR code to the website, its address.
\newcommand*{\onlineurl}{https://zpevnik.hluchnikovi.cz/}
\newcommand*{\onlineaddress}{zpevnik.hluchnikovi.cz}
\newcommand*{\booktitle}{Ještě mi chvilku zpívej}
\newcommand*{\qrheight}{20mm}
\newcommand*{\online}{%
  \begin{minipage}{100mm}\centering\color{ink}
  {\addfontfeature{LetterSpace=8}\scshape celý zpěvník}\enspace{\large\itshape\booktitle}\par
  \vspace{5mm}
  \qrcode[height=\qrheight, nolink]{\onlineurl}\par
  \vspace{4.5mm}
  {\addfontfeature{LetterSpace=10}\scshape\onlineaddress}\par
  \end{minipage}}
% A small scout lily between two thin rules.
\newcommand*{\lilyrule}{{\color{ink}%
  \rule[0.6ex]{14mm}{0.3pt}\enspace{\symbolfont\Large ⚜}\enspace\rule[0.6ex]{14mm}{0.3pt}}}

% (x, y) in mm from the trim's lower left corner T.
\newcommand*{\at}[2]{([shift={({(#1)*1mm},{(#2)*1mm})}]T)}
\newcommand*{\middlex}{\trimwidth/2}
% The whole bleed box, and a band of it from the bottom of the bleed up to #1 mm.
\newcommand*{\bleedbox}{\at{-\bleed}{-\bleed} rectangle \at{\trimwidth+\bleed}{\trimheight+\bleed}}
\newcommand*{\bottomband}[1]{\at{-\bleed}{-\bleed} rectangle \at{\trimwidth+\bleed}{#1}}
\newcommand*{\paper}[1]{\node[inner sep=0pt, anchor=south west] at \at{-\bleed}{-\bleed}
  {\includegraphics[width=\numexpr\trimwidth+2*\bleed\relax mm]{assets/paper-#1.jpg}};}
% A row of spruces (assets/spruces-#2.tex) in colour #1, its bottom #3 mm above the trim's.
\newcommand*{\spruces}[3]{\begin{scope}[shift={(T)}, yshift=#3 mm, #1]\input{assets/spruces-#2}\end{scope}}

% The designs. The text sits in the lower third, in the optical middle, or above the trees.
\newcommand*{\lowerthird}{28}
\newcommand{\backdesign}[1]{%
  \ifcsdef{back@#1}{\csuse{back@#1}}{\PackageError{back}{No back design `#1'}{}}}
% Forest: two rows of spruces over the lower part, the text in the sky.
\csdef{back@forest}{\paper{cream}
  \spruces{farspruce}{far}{18}
  \fill[farspruce] \bottomband{45};
  \spruces{spruce}{near}{-\bleed}
  \fill[spruce] \bottomband{12};
  \node at \at{\middlex}{150} {\online};}
% Low row: small spruces along the bottom edge.
\csdef{back@row}{\paper{cream}
  \spruces{spruce}{row}{-\bleed}
  \fill[spruce] \bottomband{-0.5};
  \node[anchor=south] at \at{\middlex}{44} {\online};}
% Lily: the lily between two rules over the text, on pale green paper.
\csdef{back@lily}{\paper{sage}
  \node at \at{\middlex}{108} {\begin{minipage}{100mm}\centering\lilyrule\par
    \vspace{9mm}\online\end{minipage}};}
% Moss: deeper green paper and the text.
\csdef{back@moss}{\paper{moss}\node[anchor=south] at \at{\middlex}{\lowerthird} {\online};}
% Paper: cream paper and the text.
\csdef{back@paper}{\paper{cream}\node[anchor=south] at \at{\middlex}{\lowerthird} {\online};}
% Frame: a thin double frame like the back of an old book, the text at its foot.
\csdef{back@frame}{\paper{cream}
  \draw[ink, line width=0.4pt] \at{9}{9} rectangle \at{\trimwidth-9}{\trimheight-9};
  \draw[ink, line width=0.2pt] \at{10.4}{10.4} rectangle \at{\trimwidth-10.4}{\trimheight-10.4};
  \node[anchor=south] at \at{\middlex}{22} {\online};}
