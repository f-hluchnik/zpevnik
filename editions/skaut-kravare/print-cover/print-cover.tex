% The cover of skaut-kravare for the printer: four A5 pages for a ring binding.
%   1 the front: the drawing with its rough painted edges
%   2 the inside of the front: white
%   3 the inside of the back: white
%   4 the back: the way to the online songbook (design.tex)
% Each page has the bleed and crop marks around the trim; the PDF names the trim box and the
% bleed box. Build with XeLaTeX (README.md).
\documentclass{article}
\input{design}

% The crop marks are the book's (tools/print_pdf.py): the crop package's `cam' marks round the A5
% page, on a PDF page with the same room on every side (CROP_MARK_ROOM_MM, tools/print_layout.py). They start 5 pt from the trim; the
% pictures, drawn over them, cover the part inside the bleed.
\newcommand*{\markroom}{15}
\usepackage[paperwidth=\trimwidth mm, paperheight=\trimheight mm, margin=0pt]{geometry}
\usepackage[cam, noinfo, center, width=\numexpr\trimwidth+2*\markroom\relax mm,
            height=\numexpr\trimheight+2*\markroom\relax mm]{crop}
\pagestyle{empty}

% The trim box and the bleed box of the current page, in bp from the PDF page's lower left corner.
\newcommand*{\mmtobp}[1]{\fpeval{round((#1)*72/25.4, 3)}}
\newcommand*{\pdfbox}[2]{% #1 its name, #2 its margin from the trim (mm; negative outside)
  /#1 [\mmtobp{\markroom+#2} \mmtobp{\markroom+#2}
       \mmtobp{\markroom+\trimwidth-#2} \mmtobp{\markroom+\trimheight-#2}]}
\newcommand*{\pageboxes}{%
  \special{pdf:put @thispage << \pdfbox{TrimBox}{0} \pdfbox{BleedBox}{-\bleed} >>}}

% A page: #1 drawn from the trim's corner T, and the PDF boxes.
\newcommand{\coverpage}[1]{%
  \begin{tikzpicture}[remember picture, overlay]
    \coordinate (T) at ([shift={(-\trimwidth mm/2, -\trimheight mm/2)}]current page.center);
    #1
  \end{tikzpicture}\pageboxes\mbox{}\newpage}

\begin{document}
\coverpage{\front}
\coverpage{}
\coverpage{}
\coverpage{\back}
\end{document}
