% The cover's design, read by print-cover.tex and preview.tex (XeLaTeX): the front, the back,
% their type and colours. \front and \back draw a page inside a TikZ picture whose
% coordinate T is the trim's lower left corner; both cover the bleed too.
\usepackage{fontspec}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{tikz}
\usepackage{qrcode}

% The page (mm): A5, and the bleed the printer cuts off.
\newcommand*{\trimwidth}{148}
\newcommand*{\trimheight}{210}
\newcommand*{\bleed}{3}

% Alegreya and FreeSerif (for the lily) come with TeX Live (packages alegreya, gnu-freefont).
\setmainfont{Alegreya}[Extension=.otf, UprightFont=*-Regular, ItalicFont=*-Italic,
  SmallCapsFont=AlegreyaSC-Regular]
\newfontfamily\symbolfont{FreeSerif.otf}

\definecolor{ink}{HTML}{2E261E} % the text and the QR code: dark brown, like the drawn title

% (x, y) in mm from the trim's lower left corner T.
\newcommand*{\at}[2]{([shift={({(#1)*1mm},{(#2)*1mm})}]T)}
\newcommand*{\middlex}{\trimwidth/2}
% A picture of the bleed box (assets/, make_assets.py) over the whole bleed box.
\newcommand*{\bleedpicture}[1]{\node[inner sep=0pt, anchor=south west] at \at{-\bleed}{-\bleed}
  {\includegraphics[width=\numexpr\trimwidth+2*\bleed\relax mm]{assets/#1.jpg}};}

% The front: the drawing with its rough painted edges, on white.
\newcommand*{\front}{\bleedpicture{front}}

% The back: the way to the online songbook on beige paper, under the scout lily between two
% thin rules: "celý zpěvník" and the book's title, a QR code to the website, its address.
\newcommand*{\onlineurl}{https://zpevnik.hluchnikovi.cz/}
\newcommand*{\onlineaddress}{zpevnik.hluchnikovi.cz}
\newcommand*{\booktitle}{Ještě mi chvilku zpívej}
\newcommand*{\qrheight}{20mm}
\newcommand*{\lilyrule}{%
  \rule[0.6ex]{14mm}{0.3pt}\enspace{\symbolfont\Large ⚜}\enspace\rule[0.6ex]{14mm}{0.3pt}}
\newcommand*{\online}{%
  \begin{minipage}{100mm}\centering\color{ink}
  \lilyrule\par
  \vspace{9mm}
  {\addfontfeature{LetterSpace=8}\scshape celý zpěvník}\enspace{\large\itshape\booktitle}\par
  \vspace{5mm}
  \qrcode[height=\qrheight, nolink]{\onlineurl}\par
  \vspace{4.5mm}
  {\addfontfeature{LetterSpace=10}\scshape\onlineaddress}\par
  \end{minipage}}
\newcommand*{\onlinebottom}{28}% mm from the trim's bottom edge: the lower third
\newcommand*{\back}{\bleedpicture{paper-beige}
  \node[anchor=south] at \at{\middlex}{\onlinebottom} {\online};}
