% The cover of skaut-kravare, read by make print (tools/print_cover.py; docs/printing.md).
% The front: Klára Horká's drawing with its rough painted edges, on white (front.jpg). The insides
% stay white. The back: the way to the online songbook on beige paper (paper.jpg), under the scout
% lily between two thin rules: "celý zpěvník" and the book's title, a QR code to the website and
% its address. The text, the rules, the lily and the QR code are in black ink alone, so they print
% sharp, with no inks to register. The pictures are made from the scan by source/make_ingredients.py.
\usepackage[T1]{fontenc}
\usepackage{Alegreya}  % TeX Live's alegreya
\usepackage{microtype}
\usepackage{qrcode}

\definecolor{coverink}{cmyk}{0,0,0,1}

% A picture of the bleed box, over the whole bleed box. The pictures are made for a 3 mm bleed.
\newcommand*{\bleedpicture}[1]{%
  \node[inner sep=0pt, anchor=south west] at ([shift={(-\zpcoverbleed, -\zpcoverbleed)}]zpcovertrim)
    {\includegraphics[width=\dimexpr\zpcovertrimwidth+2\zpcoverbleed\relax]{#1}};}

\newcommand*{\zpcoverfront}{\bleedpicture{front.jpg}}

\newcommand*{\onlineurl}{https://zpevnik.hluchnikovi.cz/}
\newcommand*{\onlineaddress}{zpevnik.hluchnikovi.cz}
\newcommand*{\booktitle}{Ještě mi chvilku zpívej}
\newcommand*{\qrheight}{20mm}
% The lily (lily.pdf, from source/lily.tex) on the baseline, as the glyph it was made from.
\newcommand*{\lilydepth}{2.088pt}
\newcommand*{\lily}{\raisebox{-\lilydepth}{\includegraphics{lily.pdf}}}
\newcommand*{\lilyrule}{\rule[0.6ex]{14mm}{0.3pt}\enspace\lily\enspace\rule[0.6ex]{14mm}{0.3pt}}
\newcommand*{\online}{%
  \begin{minipage}{100mm}\centering\color{coverink}
  \lilyrule\par
  \vspace{9mm}
  {\scshape\textls[80]{celý zpěvník}}\enspace{\large\itshape\booktitle}\par
  \vspace{5mm}
  \qrcode[height=\qrheight, nolink]{\onlineurl}\par
  \vspace{4.5mm}
  {\scshape\textls[100]{\onlineaddress}}\par
  \end{minipage}}
\newcommand*{\onlinebottom}{28mm}% from the trim's bottom edge: the lower third
\newcommand*{\zpcoverback}{\bleedpicture{paper.jpg}
  \node[anchor=south] at ([shift={(0.5\zpcovertrimwidth, \onlinebottom)}]zpcovertrim) {\online};}
