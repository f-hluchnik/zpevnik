\hyt{marienavarova}
\song{Marie} \interpret{zuzananavarova}{Zuzana Navarová}

\vers{1}{
Marie \chord{Hm}má se vracet, \chord{F\kk\7}ta co tu \chord{Hm}bydlí. Marie \chord{Hm}má se vracet, \chord{H\7}tak postav \chord{Em}židli.\\
Marie \chord{Em}bílý racek, Marie \chord{Hm}má se vracet, ty si tu \chord{C\kk\7}dáváš dvacet, tak abys \chord{F\kk\7}vstal.
}

\vers{2}{
Marie má se vracet, co by ses divil?\\
Marie má se vracet, ta, cos ji mydlil.\\
Marie modrý ptáček, Marie moudivláček,\\
s kufříkem od natáček, ta, cos ji štval.
}

\refrain{
Marie, \chord{Hm}hoď sem cihlu, má se vracet, trá \chord{Em}ratata!
}

\vers{3}{
Marie má se vracet, píšou, že lehce.\\
Marie má se vracet, že už tě nechce.\\
Ta, co jí není dvacet, Marie má se vracet,\\
ta, cos jí dal pár facek, a pak s ní spal.
} \refsm{}

\vers{4}{
Marie má se vracet, ta, co tu bydlí.\\
Marie má se vracet, tak postav židli.\\
Olej a těžký kola, vlak někam do Opola,\\
herbatka, jedna Cola, no tak se sbal.
} \refsm{}

\bridge{
Marie \chord{Em}holubice, Marie \chord{Hm}létavice,\\
Marie \chord{C\kk m}blýskavice, krasavice, \chord{F\kk 7}ech, Rosice -- Pardubice.
}

\vers{5}{
Má se vracet, ta, co tu bydlí.\\
Marie má se vracet, tak postav židli.\\
Marie bílý racek, Marie má se vracet,\\
ty si tu dáváš dvacet, tak abys vstal.
}

\vers{6}{
Marie má se vracet, co by ses divil?\\
Marie má se vracet, ta, cos ji mydlil.\\
Marie modrý ptáček, Marie moudivláček,\\
tak postav \chord{Em}židli a \chord{F\kk\7}sbal svý fid\chord{Hm}li.
}
