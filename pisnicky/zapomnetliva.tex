\song{Zapomnětlivá} \artist{uhlirsverak}
\alttitle{Takovej ten s takovou tou}

\vers[1]{
\lbl{A:}\chord{Am}Kde to je? Teď před chvílí jsem \chord{E\7}to tu ještě \chord{Am}měl.\\
\lbl{B:}\chord{Am}Co hledáš, švagře milý? Možná, že \chord{E\7}bych vě\chord{Am}děl.\\
\lbl{A:}\rep{\chord{C}Takovej ten, s \chord{G}takovou tou, \chord{F}jak to jen říci \chord{C}jednou větou?\\

\lbl{B:}\chord{F}Bohužel, švagře \chord{E\7}nerozu\chord{Am}mím.}

\vers[2]{
\lbl{A:}No tohle, to jsem blázen, to jsou mi problémy.\\
\lbl{B:}Možná to spadlo na zem a je to na zemi.\\
\lbl{A:}\rep{Na ze mi ne, to bych si všim', proč já se trochu nesoustředím?\\
\nv{}Teď jsem to měl, a je to pryč.\\
\lbl{B:}Tak mi to, švagře, trochu vylič.}
}

\bridge{
\lbl{A:}\chord{A}Nahoře to má, jako, takový \chord{Dm}čudlíky.\\
\lbl{B:}\chord{H\7}Ach, švagře, ty jsi pako. \chord{E\7}Jak je to veliký?\\
\lbl{A:}\chord{G}Podívej,
}

\vers[3]{
\lbl{A:}\rep{Bejt to velký, a ne malý, to bysme to tak nehledali.\\
\lbl{B:}To jsou trable, takhle k stáru, třebas' to zakop' pod almaru.}
}

\vers[4]{
\lbl{A:}Tam jsem byl, pod tou skříní, tam jsou jsou jen pavouci.\\
\lbl{B:}Řeknu ti, švagře, nyní tu pravdu pravdoucí.\\
\nv{}Jak dobře víš, s tebou je kříž, ty tvoje věci maj' někde skrýš.\\
\nv{}Jelikož jsou ale němé, tak už je nikdy nenajdeme.
}

\vers[5]{
\rep{Takovej ten, s takovou tou, jak to jen říci jednou větou?\\
Taková ta, s takovým tím,\\
\lbl{A:}Já tomu, švagře, nerozumím.}
}

\coda{Kde to je, teď před chvílí jsem to tu ještě měl.}
